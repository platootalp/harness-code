## 1. 概述

### 1.1 模块名称

Memory（记忆系统）

### 1.2 职责

Memory 模块是 Context 模块的底层存储支撑，属于**工具提供者（Tool Provider）**角色：

- 管理短期记忆（滑动窗口内的最近对话内容）
- 管理长期记忆（语义、情景、程序三种记忆）
- 通过 Retriever 接口为 Context 提供记忆召回能力
- **不直接参与上下文压缩**——由 Context 的 WindowManager 统一管理

### 1.3 核心能力

| 能力 | 说明 |
| ---- | ---- |
| 短期记忆管理 | 基于滑动窗口的最近 N 轮对话，自动淘汰 |
| 长期记忆存储 | 三种记忆类型，支持向量检索 |
| 多后端支持 | 向量存储、Embedding 模型可插拔 |
| 混合检索 | 向量相似度 + metadata 标量过滤 |
| 记忆持久化 | 跨会话恢复 |

---

## 2. 核心问题与解决方案

### 2.1 记忆容量限制

**问题描述**：向量存储有容量限制，长期记忆持续积累会耗尽空间。

**解决方案**：ShortTermMemory 滑动窗口自动淘汰；长期记忆由 Context WindowManager 阈值触发 Compress 策略。

### 2.2 检索质量

**问题描述**：如何从大量记忆中找到最相关的内容。

**解决方案**：向量语义检索 + memory_type 标量过滤，支持混合检索。

### 2.3 短期与长期记忆协同

**问题描述**：两种记忆有不同的管理策略，需要统一协调。

**解决方案**：MemoryRetriever 统一协调召回，短期直接返回，长期向量检索。

## 3. 数据模型与状态机

### 3.1 核心类型定义

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Any
from uuid import uuid4


class MemoryType(Enum):
    """记忆类型"""
    SHORT_TERM = "short_term"    # 短期记忆 - 滑动窗口
    SEMANTIC = "semantic"        # 语义记忆 - 一般知识
    EPISODIC = "episodic"        # 情景记忆 - 过去事件
    PROCEDURAL = "procedural"     # 程序记忆 - 行为模式


class SessionStatus(Enum):
    """会话状态"""
    ACTIVE = "active"
    IDLE = "idle"
    ARCHIVED = "archived"
    EXPIRED = "expired"


@dataclass
class MemoryBlock:
    """记忆块"""
    id: str = field(default_factory=lambda: str(uuid4()))
    session_id: str = ""
    content: str = ""
    memory_type: MemoryType = MemoryType.SEMANTIC
    embedding: list[float] | None = None
    importance: float = 0.5
    status: SessionStatus = SessionStatus.ACTIVE
    created_at: datetime = field(default_factory=datetime.now)
    accessed_at: datetime = field(default_factory=datetime.now)
    metadata: dict[str, Any] = field(default_factory=dict)


@dataclass
class MemoryConfig:
    """记忆配置"""
    vector_store: str = "qdrant"           # 向量存储后端
    embedding_model: str = "text-embedding-3-small"
    embedding_dimension: int = 1536
    short_term_window_size: int = 10
    short_term_max_tokens: int = 8000
    retrieval_limit: int = 10
    similarity_threshold: float = 0.7
    duplication_threshold: float = 0.85     # 合并阈值，超过则合并
```

### 3.2 记忆类型

| 类型 | 用途 | 存储方式 |
| ---- | ---- | -------- |
| **语义记忆（Semantic）** | 一般知识和事实 | 向量存储 |
| **情景记忆（Episodic）** | 具体过去事件和交互 | 向量存储 |
| **程序记忆（Procedural）** | 行为模式和操作知识 | 向量存储 |

from abc import ABC, abstractmethod


class VectorStore(ABC):
    """向量存储抽象接口"""

    @abstractmethod
    async def upsert(self, blocks: list[MemoryBlock]) -> None:
        """插入或更新记忆"""

    @abstractmethod
    async def search(
        self,
        query_embedding: list[float],
        filters: dict[str, Any] | None = None,
        limit: int = 10,
        similarity_threshold: float = 0.7,
    ) -> list[tuple[MemoryBlock, float]]:
        """向量检索，返回 (记忆块, 相似度) 列表"""

    @abstractmethod
    async def delete(self, memory_id: str) -> None:
        """删除记忆"""

### 5.3 Memory 协调器（完整 CRUD + 冲突解决）

```python
class Memory:
    """Memory 协调器 - 完整增删改查 + 冲突解决"""

    def __init__(
        self,
        short_term: ShortTermMemory,
        long_term: LongTermMemory,
        retriever: MemoryRetriever,
        embedder: Embedder,
        config: MemoryConfig,
    ) -> None:
        self._short_term = short_term
        self._long_term = long_term
        self._retriever = retriever
        self._embedder = embedder
        self._config = config

    # ==================== 增删改查 ====================

    async def add(
        self,
        session_id: str,
        content: str,
        memory_type: MemoryType,
        metadata: dict[str, Any] | None = None,
    ) -> MemoryBlock:
        """
        添加记忆（自动去重）

        冲突解决策略：
        1. 计算与已有记忆的相似度
        2. 超过阈值 → 合并到已有记忆
        3. 未超过阈值 → 创建新记忆
        """
        if memory_type == MemoryType.SHORT_TERM:
            return await self._short_term.add(session_id, content, metadata)

        # 长期记忆：检查是否需要合并
        embedding = await self._embedder.embed([content])[0]
        existing = await self._long_term.search(
            query_embedding=embedding,
            session_id=session_id,
            memory_types=[memory_type],
            limit=1,
            similarity_threshold=self._config.duplication_threshold,
        )

        if existing:
            # 合并到已有记忆
            return await self._merge(existing[0], content, metadata)
        else:
            # 创建新记忆
            return await self._long_term.add(session_id, content, memory_type, metadata)

    async def update(
        self,
        memory_id: str,
        content: str | None = None,
        metadata: dict[str, Any] | None = None,
    ) -> MemoryBlock | None:
        """更新记忆（仅长期记忆）"""
        return await self._long_term.update(memory_id, content, metadata)

    async def delete(self, memory_id: str, session_id: str) -> bool:
        """删除记忆"""
        memory_type = await self._long_term.get_memory_type(memory_id)
        if memory_type == MemoryType.SHORT_TERM:
            return await self._short_term.delete(memory_id)
        return await self._long_term.delete(memory_id, session_id)

    async def get(self, memory_id: str, session_id: str) -> MemoryBlock | None:
        """获取指定记忆"""
        return await self._long_term.get(memory_id, session_id)

    async def get_all(
        self,
        session_id: str,
        memory_type: MemoryType | None = None,
        limit: int = 100,
    ) -> list[MemoryBlock]:
        """获取会话所有记忆"""
        return await self._long_term.get_all(session_id, memory_type, limit)

    async def history(self, memory_id: str) -> list[MemoryBlock]:
        """获取记忆的变更历史"""
        return await self._long_term.get_history(memory_id)

    # ==================== 召回 ====================

    async def recall(
        self,
        query: str,
        session_id: str,
        memory_types: list[MemoryType] | None = None,
        limit: int = 10,
    ) -> list[MemoryBlock]:
        """召回相关记忆"""
        return await self._retriever.recall(query, session_id, memory_types, limit)

    # ==================== 冲突解决 ====================

    async def _merge(
        self,
        existing: MemoryBlock,
        new_content: str,
        metadata: dict[str, Any] | None = None,
    ) -> MemoryBlock:
        """
        合并记忆

        策略：
        1. 更新 importance = max(existing, new)
        2. 合并 metadata
        3. 更新时间戳
        4. 追加内容到 memory_notes
        """
        updated = MemoryBlock(
            id=existing.id,
            session_id=existing.session_id,
            content=existing.content,
            memory_type=existing.memory_type,
            embedding=existing.embedding,
            importance=max(existing.importance, (metadata.get("importance", 0.5) if metadata else 0.5)),
            created_at=existing.created_at,
            accessed_at=datetime.now(),
            metadata={
                **existing.metadata,
                **(metadata or {}),
                "merged_from": new_content,
                "merge_count": existing.metadata.get("merge_count", 0) + 1,
            },
        )
        await self._long_term.update_block(updated)
        return updated
```

### 5.4 冲突解决策略

**合并策略**：

| 场景 | 策略 |
| ---- | ---- |
| 相似度 > 阈值 | 合并到已有记忆 |
| 相似度 <= 阈值 | 创建新记忆 |
| 完全相同内容 | 跳过，返回已有 |

**元数据合并**：
- 保留已有 metadata
- 增量追加新 metadata
- 记录 merge_count 和 merged_from

### 5.5 模块上下文交互

| 层级 | 与 Memory 的交互 |
| ---- | ---------------- |
| Context | 调用 add() 存储，recall() 召回 |
| Session | 提供会话历史作为记忆来源 |
| VectorStore | 提供向量存储能力（可插拔） |

## 7. 实现细节

### 7.1 短期记忆滑动窗口

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any
from uuid import uuid4

from .models import Message, MessageRole


class ShortTermMemory:
    """短期记忆 - 内存存储，滑动窗口"""

    def __init__(
        self,
        session_id: str,
        window_size: int = 10,
        max_tokens: int = 8000,
    ) -> None:
        self._session_id = session_id
        self._window_size = window_size
        self._max_tokens = max_tokens
        self._messages: list[Message] = []
        self._token_count: int = 0

    async def add(
        self,
        session_id: str,
        content: str,
        metadata: dict[str, Any] | None = None,
    ) -> MemoryBlock:
        """追加消息到滑动窗口"""
        message = Message(
            id=str(uuid4()),
            session_id=session_id,
            content=content,
            role=MessageRole.USER,
            metadata=metadata or {},
        )
        self._messages.append(message)
        self._token_count += self._estimate_tokens(content)
        await self._evict_if_needed()
        return MemoryBlock(
            id=message.id,
            session_id=session_id,
            content=content,
            memory_type=MemoryType.SHORT_TERM,
        )

    async def get_recent(self, n: int | None = None) -> list[MemoryBlock]:
        """获取最近 n 条记忆"""
        messages = self._messages[-n:] if n else list(self._messages)
        return [
            MemoryBlock(
                id=m.id,
                session_id=m.session_id,
                content=m.content,
                memory_type=MemoryType.SHORT_TERM,
            )
            for m in messages
        ]

    async def truncate_to(self, max_messages: int) -> list[MemoryBlock]:
        """截断到指定消息数"""
        if len(self._messages) <= max_messages:
            return []
        removed = self._messages[:len(self._messages) - max_messages]
        self._messages = self._messages[-max_messages:]
        return [
            MemoryBlock(
                id=m.id,
                session_id=m.session_id,
                content=m.content,
                memory_type=MemoryType.SHORT_TERM,
            )
            for m in removed
        ]

    async def _evict_if_needed(self) -> None:
        """滑动窗口淘汰"""
        while (
            len(self._messages) > self._window_size
            or self._token_count > self._max_tokens
        ) and self._messages:
            removed = self._messages.pop(0)
            self._token_count -= self._estimate_tokens(removed.content)

    def _estimate_tokens(self, content: str) -> int:
        return len(content) // 4
```

class VectorStore(ABC):
    """向量存储抽象接口"""

    @abstractmethod
    async def upsert(self, blocks: list[MemoryBlock]) -> None:
        """插入或更新记忆"""

        """向量检索"""

    @abstractmethod
    async def delete(self, memory_id: str) -> None:
        """删除记忆"""


class VectorStoreFactory:
    """向量存储工厂"""

    _stores: dict[str, type[VectorStore]] = {}

    @classmethod
    def register(cls, name: str, store_class: type[VectorStore]) -> None:
        cls._stores[name] = store_class

    @classmethod
    def create(cls, name: str, **kwargs) -> VectorStore:
        if name not in cls._stores:
            raise ValueError(f"Unknown vector store: {name}")
        return cls._stores[name](**kwargs)

### 7.3 长期记忆实现

```python
class LongTermMemory:
    """长期记忆 - 向量存储封装"""

    def __init__(
        self,
        vector_store: VectorStore,
        embedder: Embedder,
    ) -> None:
        self._store = vector_store
        self._embedder = embedder
        self._blocks: dict[str, MemoryBlock] = {}  # 内存缓存
        self._history: dict[str, list[MemoryBlock]] = {}  # 变更历史

    async def add(
        self,
        session_id: str,
        content: str,
        memory_type: MemoryType,
        metadata: dict[str, Any] | None = None,
    ) -> MemoryBlock:
        """添加记忆"""
        embedding = await self._embedder.embed([content])[0]
        block = MemoryBlock(
            id=str(uuid4()),
            session_id=session_id,
            content=content,
            memory_type=memory_type,
            embedding=embedding,
            metadata=metadata or {},
        )
        self._blocks[block.id] = block
        await self._store.upsert([block])
        return block

    async def search(
        self,
        query_embedding: list[float],
        session_id: str,
        memory_types: list[MemoryType] | None = None,
        limit: int = 10,
        similarity_threshold: float = 0.7,
    ) -> list[MemoryBlock]:
        """语义检索"""
        filters = {"session_id": session_id}
        if memory_types:
            filters["memory_type"] = [t.value for t in memory_types]

        results = await self._store.search(
            query_embedding=query_embedding,
            filters=filters,
            limit=limit,
            similarity_threshold=similarity_threshold,
        )
        return [block for block, _ in results]

    async def update(
        self,
        memory_id: str,
        content: str | None = None,
        metadata: dict[str, Any] | None = None,
    ) -> MemoryBlock | None:
        """更新记忆"""
        if memory_id not in self._blocks:
            return None

        block = self._blocks[memory_id]

        # 记录历史
        self._history.setdefault(memory_id, []).append(block)

        # 更新字段
        if content is not None:
            block.content = content
            block.embedding = await self._embedder.embed([content])[0]
        if metadata is not None:
            block.metadata = {**block.metadata, **metadata}
        block.accessed_at = datetime.now()

        await self._store.upsert([block])
        return block

    async def update_block(self, block: MemoryBlock) -> None:
        """直接更新记忆块"""
        self._history.setdefault(block.id, []).append(self._blocks.get(block.id))
        self._blocks[block.id] = block
        await self._store.upsert([block])

    async def delete(self, memory_id: str) -> bool:
        """删除记忆"""
        if memory_id in self._blocks:
            del self._blocks[memory_id]
            await self._store.delete(memory_id)
            return True
        return False

    async def get(self, memory_id: str, session_id: str) -> MemoryBlock | None:
        """获取指定记忆"""
        if memory_id in self._blocks:
            block = self._blocks[memory_id]
            if block.session_id == session_id:
                return block
        return None

    async def get_all(
        self,
        session_id: str,
        memory_type: MemoryType | None = None,
        limit: int = 100,
    ) -> list[MemoryBlock]:
        """获取会话所有记忆"""
        blocks = [
            b for b in self._blocks.values()
            if b.session_id == session_id
            and (memory_type is None or b.memory_type == memory_type)
        ]
        blocks.sort(key=lambda b: b.accessed_at, reverse=True)
        return blocks[:limit]

    async def get_history(self, memory_id: str) -> list[MemoryBlock]:
        """获取记忆的变更历史"""
        return self._history.get(memory_id, [])
```

### 7.4 记忆检索器

```python
class MemoryRetriever:
    """记忆检索器 - 协调短/长期记忆"""

    def __init__(
        self,
        short_term: ShortTermMemory,
        long_term: LongTermMemory,
        embedder: Embedder,
        config: MemoryConfig,
    ) -> None:
        self._short_term = short_term
        self._long_term = long_term
        self._embedder = embedder
        self._config = config

    async def recall(
        self,
        query: str,
        session_id: str,
        memory_types: list[MemoryType] | None = None,
        limit: int = 10,
    ) -> list[MemoryBlock]:
        """召回相关记忆"""
        results: list[MemoryBlock] = []

        # 1. 短期记忆 - 直接返回
        if memory_types is None or MemoryType.SHORT_TERM in memory_types:
            recent = await self._short_term.get_recent(n=limit)
            results.extend(recent)

        # 2. 长期记忆 - 语义检索
        lt_types = memory_types or [
            MemoryType.SEMANTIC,
            MemoryType.EPISODIC,
            MemoryType.PROCEDURAL,
        ]
        lt_types = [t for t in lt_types if t != MemoryType.SHORT_TERM]
        if lt_types:
            query_embedding = await self._embedder.embed([query])[0]
            lt_results = await self._long_term.search(
                query_embedding=query_embedding,
                session_id=session_id,
                memory_types=lt_types,
                limit=limit,
            )
            results.extend(lt_results)

        return results[:limit]
```

### 7.5 Embedder 工厂

```python
from abc import ABC, abstractmethod

import openai


class Embedder(ABC):
    """Embedding 抽象接口"""

    @abstractmethod
    async def embed(self, texts: list[str]) -> list[list[float]]:
        """批量生成 embedding"""


class OpenAIEmbedder(Embedder):
    """OpenAI Embedder"""

    def __init__(self, model: str = "text-embedding-3-small") -> None:
        self._model = model

    async def embed(self, texts: list[str]) -> list[list[float]]:
        # 实际调用 OpenAI API
        response = await openai.Embedding.acreate(
            model=self._model,
            input=texts,
        )
        return [item["embedding"] for item in response["data"]]


class EmbedderFactory:
    """Embedder 工厂"""

    _embedders: dict[str, type[Embedder]] = {}

    @classmethod
    def register(cls, name: str, embedder_class: type[Embedder]) -> None:
        cls._embedders[name] = embedder_class

    @classmethod
    def create(cls, name: str, **kwargs) -> Embedder:
        if name not in cls._embedders:
            raise ValueError(f"Unknown embedder: {name}")
        return cls._embedders[name](**kwargs)


# 注册默认 embedder
EmbedderFactory.register("openai", OpenAIEmbedder)
```

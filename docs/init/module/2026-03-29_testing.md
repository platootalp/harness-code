## 5. 核心测试用例

### 5.1 Orchestrator 测试

**任务规约解析测试**：

```python
@pytest.mark.unit
class TestTaskSpecParser:
    async def test_task_spec_goal_extraction(self):
        """测试任务目标提取"""
        parser = TaskSpecParser()
        spec = await parser.parse("帮我写一个快速排序函数", context={})
        assert spec.goal == "帮我写一个快速排序函数"

    async def test_task_spec_clarification_triggers(self):
        """测试需要澄清的场景"""
        parser = TaskSpecParser()
        spec = await parser.parse("帮我优化一下", context={})
        assert spec.requires_clarification is True
```

### 5.2 Config 测试

```python
@pytest.mark.unit
class TestConfigLoader:
    async def test_load_success(self, schema: dict, valid_config: Path) -> None:
        """测试成功加载配置"""
        loader = ConfigLoader(schema=schema)
        config = await loader.load(valid_config)
        assert config.version == "1.0"

    async def test_load_file_not_found(self, schema: dict) -> None:
        """测试配置文件不存在"""
        loader = ConfigLoader(schema=schema)
        with pytest.raises(ConfigNotFoundError):
            await loader.load("nonexistent.json")


@pytest.mark.unit
class TestEnvInjection:
    def test_inject_simple_var(self, monkeypatch: pytest.MonkeyPatch) -> None:
        """测试简单变量注入"""
        monkeypatch.setenv("TEST_API_KEY", "secret123")
        config = {"api_key": "${TEST_API_KEY}"}
        result = inject_env_vars(config)
        assert result["api_key"] == "secret123"
```

### 5.3 Tools 测试

```python
@pytest.mark.unit
class TestReadFileTool:
    async def test_read_file_success(self, tmp_path: Path):
        """测试成功读取文件"""
        test_file = tmp_path / "test.txt"
        test_file.write_text("Hello, World!")
        tool = ReadFileTool()
        context = ToolContext(working_directory=str(tmp_path))
        result = await tool.execute(context, path="test.txt")
        assert result.success is True
        assert result.output == "Hello, World!"

    async def test_read_file_not_found(self, tmp_path: Path):
        """测试文件不存在"""
        tool = ReadFileTool()
        context = ToolContext(working_directory=str(tmp_path))
        result = await tool.execute(context, path="nonexistent.txt")
        assert result.success is False
```

### 5.4 Session 测试

```python
@pytest.mark.unit
class TestSessionManager:
    async def test_create_session(self, manager: SessionManager) -> None:
        """测试创建会话"""
        session = await manager.create(user_id="user1")
        assert session.id is not None
        assert session.metadata.user_id == "user1"
        assert session.status == SessionStatus.ACTIVE

    async def test_get_session(self, manager: SessionManager) -> None:
        """测试获取会话"""
        created = await manager.create()
        retrieved = await manager.get(created.id)
        assert retrieved is not None
        assert retrieved.id == created.id
```

### 5.5 Observability 测试

```python
@pytest.mark.unit
class TestStructuredLogger:
    def test_log_level_filter(self):
        """测试日志级别过滤"""
        records = []
        exporter = TestLogExporter(records)
        logger = StructuredLogger("test", exporter)
        logger.set_level(LogLevel.INFO)
        logger.debug("debug message")  # 应该被过滤
        logger.info("info message")
        assert len(records) == 1

    def test_json_format(self):
        """测试 JSON 格式输出"""
        records = []
        exporter = TestLogExporter(records, format=LogFormat.JSON)
        logger = StructuredLogger("test", exporter)
        logger.info("test message", foo="bar")
        assert records[0].metadata["foo"] == "bar"


@pytest.mark.unit
class TestTracer:
    def test_trace_id_propagation(self):
        """测试追踪 ID 传播"""
        tracer = Tracer("test")
        parent = tracer.start_span("parent")
        assert parent.trace_id is not None
        child = tracer.start_span("child")
        assert child.trace_id == parent.trace_id
        assert child.parent_span_id == parent.span_id
```

## 6. Mock 策略

### 6.1 通用 Mock 策略

| 依赖组件 | Mock 方式 | 说明 |
| ------- | -------- | ---- |
| Storage | AsyncMock | 模拟文件读写操作 |
| EventBus | AsyncMock | 模拟事件发布/订阅 |
| Model Adapter | httpx_mock | 模拟 HTTP 响应 |
| VectorStore | Mock | 模拟向量存储 |
| EmbeddingGenerator | Mock | 模拟 embedding 生成 |

### 6.2 测试数据管理

- 使用 fixture 管理测试会话
- Mock 外部依赖（Model Adapter、Storage）
- 使用内存数据库进行集成测试
- 测试数据与生产数据隔离

## 7. 集成测试策略

### 7.1 测试类型与执行频率

| 测试类型 | 执行频率 | 覆盖场景 |
